1)Файлы, которые нужно знать наизусть
Файл:
1.1)/etc/passwd -  Информаиця о пользователях(логин, UID, GID, домашняя папка, shell)
1.2)/etc/shadow - Хэш паролей(только root) -rw-r-----
1.3)/etc/group - Группы пользователей -rw-r--r--
2)Специальные биты(SUID,SGID, Sticky)
Бит	Обозначение	Действие	Пример
SUID (Set User ID)	rwsr-xr-x	Программа запускается от имени владельца (а не того, кто запустил)	passwd, ping
SGID (Set Group ID)	rwxr-sr-x	Программа/файл наследуют группу владельца	общие папки проекта
Sticky Bit	rwxrwxrwt	Только владелец может удалить свой файл в общей папке	/tmp
Установка специальных битов

chmod со спецбитами (цифровой формат с 4 цифрами)
bash

chmod 4755 script.sh   # SUID
chmod 2755 script.sh   # SGID  
chmod 1777 /tmp        # sticky bit

Цифра	Что значит
первая (4,2,1)	спецбиты: 4=SUID, 2=SGID, 1=sticky
вторая	права для владельца (7=rwx)
третья	права для группы
четвёртая	права для остальных

Пример: chmod 4755 = SUID + rwxr-xr-x

chmod u+s файл      # установить SUID
chmod g+s файл      # установить SGID
chmod +t папка      # установить sticky bit

# Цифровой режим (четвёртая цифра)
chmod 4755 файл     # 4 = SUID, 7=rwx, 5=r-x
chmod 2755 файл     # 2 = SGID
chmod 1777 папка    # 1 = sticky bit

Поиск файлов с SUID - find / -perm -4000 -type f 2>/dev/null

3)Управление пользователями и группами

#Создание/удаление пользователя
sudo useradd -m -s /bin/bash bob # Создание пользователя + его домашнюю папку
sudo passwd bob # Установка пароля
sudo userdel -r bob # Удаления пользователя и уго файлы

#Создание/удаление группы
sudo groupadd dev # создание группы
sudo usermod -aF dev bob #Добавление пользователя в группу
sudo gpasswd -d bob dev # Удаление Пользователя из группы

#Информация о пользователе
id bob
groups bob
last #последние логи

Смена Владельца и группы
chown bob файл                 # сменить владельца
chown bob:developers файл      # сменить владельца и группу
chown :developers файл         # сменить только группу
chgrp developers файл          # тоже сменить группу
chown -R bob папка/            # рекурсивно


📌 Короткая шпаргалка по флагам (для запоминания)
Команда	Флаг	Что делает
useradd	-m	создать домашнюю папку
useradd	-s	указать shell
userdel	-r	удалить домашнюю папку
usermod	-aG	добавить в группу (без потери других)
chmod	4xxx	установить SUID
chmod	2xxx	установить SGID
chmod	1xxx	установить sticky bit
find	-perm	поиск по правам
find	-type f	только файлы
